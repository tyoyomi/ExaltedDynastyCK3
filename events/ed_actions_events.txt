namespace = ed_actions

# Exalted Dynasty Actions hub — Done first (SS/EC craft).
# Former major verbs: exalt / revoke / mass sync / auto-strip leavers.
# Event options hide when gated (engine has no grey-out for event options).

ed_actions.1001 = {
	type = character_event
	title = ed_actions.1001.t
	theme = dynasty

	desc = ed_actions.1001.desc

	# Done first — closes hub
	option = {
		name = ed_actions.1001.close
		custom_tooltip = ed_actions.tt.close
	}

	# Exalt My House
	option = {
		name = ed_actions.1001.exalt
		trigger = {
			exists = house
			NOT = { has_game_rule = ed_rule_granting_disabled }
			OR = {
				is_ai = no
				has_game_rule = ed_rule_granting_player_and_ai
			}
			NOT = { ed_character_house_is_exalted = yes }
			custom_tooltip = {
				text = ed_tt_actor_may_exalt
				ed_actor_may_exalt = yes
			}
		}
		custom_tooltip = ed_exalt_my_house_decision_effect_tt
		ed_exalt_house_effect = yes
		trigger_event = ed_actions.1001
	}

	# Revoke House Exaltation
	option = {
		name = ed_actions.1001.revoke
		trigger = {
			exists = house
			ed_character_house_is_exalted = yes
			is_house_head = yes
			custom_tooltip = {
				text = ed_tt_actor_may_revoke
				ed_actor_may_revoke = yes
			}
		}
		custom_tooltip = ed_revoke_house_exaltation_decision_effect_tt
		ed_revoke_house_effect = yes
		trigger_event = ed_actions.1001
	}

	# Mass Sync — toast ed_events.1501 owns next click (do not re-open)
	option = {
		name = ed_actions.1001.sync
		trigger = {
			exists = house
			ed_character_house_is_exalted = yes
			NOT = { has_game_rule = ed_rule_granting_disabled }
			OR = {
				is_ai = no
				has_game_rule = ed_rule_granting_player_and_ai
			}
			custom_tooltip = {
				text = ed_tt_house_is_exalted
				ed_character_house_is_exalted = yes
			}
		}
		custom_tooltip = ed_mass_sync_exalted_house_decision_effect_tt
		ed_mass_sync_exalted_house_effect = yes
		trigger_event = ed_events.1501
	}

	# Auto-Strip Leavers — toast ed_events.1502 owns next click (do not re-open)
	option = {
		name = ed_actions.1001.strip
		trigger = {
			exists = house
			ed_character_house_is_exalted = yes
			has_game_rule = ed_rule_strip_on_leave_house_on
			OR = {
				is_ai = no
				has_game_rule = ed_rule_granting_player_and_ai
			}
		}
		custom_tooltip = ed_auto_strip_left_house_members_decision_effect_tt
		ed_auto_strip_left_house_members_effect = yes
		trigger_event = ed_events.1502
	}
}
