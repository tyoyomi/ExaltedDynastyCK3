# Exalted Dynasty — scripted triggers

ed_is_exalted_dynast = {
	has_trait = ed_exalted_dynast
}

# Soft peer: Supreme Sovereign trait key. Fails closed if SS is absent (may log once).
ed_character_is_supreme_sovereign = {
	has_trait = Supreme_Sovereign
}

# this = house
ed_house_is_exalted = {
	has_variable = ed_exalted
}

ed_house_is_secondary_exalted = {
	has_variable = ed_exalted
	has_variable = ed_exalted_secondary
}

ed_house_is_primary_exalted = {
	has_variable = ed_exalted
	NOT = { has_variable = ed_exalted_secondary }
}

# Character's current house is exalted
ed_character_house_is_exalted = {
	exists = house
	house = {
		ed_house_is_exalted = yes
	}
}

ed_character_house_is_secondary_exalted = {
	exists = house
	house = {
		ed_house_is_secondary_exalted = yes
	}
}

ed_character_house_is_primary_exalted = {
	exists = house
	house = {
		ed_house_is_primary_exalted = yes
	}
}

ed_granting_disabled = {
	has_game_rule = ed_rule_granting_disabled
}

ed_actor_may_exalt_by_granting_rule = {
	NOT = { has_game_rule = ed_rule_granting_disabled }
	trigger_if = {
		limit = { has_game_rule = ed_rule_granting_player_only }
		is_ai = no
	}
	# player_and_ai: both allowed
}

ed_actor_meets_who_may_exalt = {
	trigger_if = {
		limit = { has_game_rule = ed_rule_who_may_exalt_house_head }
		is_house_head = yes
	}
	trigger_else_if = {
		limit = { has_game_rule = ed_rule_who_may_exalt_dynasty_head }
		is_dynast = yes
	}
	trigger_else = {
		# either
		OR = {
			is_house_head = yes
			is_dynast = yes
		}
	}
}

ed_house_head_blocks_for_sovereign = {
	has_game_rule = ed_rule_stack_with_sovereign_block_house_if_head_sovereign
	exists = house
	house = {
		exists = house_head
		house_head = {
			ed_character_is_supreme_sovereign = yes
		}
	}
}

# Actor eligible to exalt their own house (actor = this) — primary path
ed_actor_may_exalt = {
	is_alive = yes
	exists = house
	ed_actor_may_exalt_by_granting_rule = yes
	ed_actor_meets_who_may_exalt = yes
	NOT = { ed_character_house_is_exalted = yes }
	custom_tooltip = {
		text = ed_tt_not_blocked_by_sovereign_head
		NOT = { ed_house_head_blocks_for_sovereign = yes }
	}
	# v1: own house only (vassal-house reserved)
	trigger_if = {
		limit = { has_game_rule = ed_rule_scope_any_vassal_house }
		always = yes
	}
}

# Current house head of an exalted house may revoke (primary or secondary)
ed_actor_may_revoke = {
	is_alive = yes
	exists = house
	is_house_head = yes
	ed_character_house_is_exalted = yes
}

ed_should_strip_member = {
	has_trait = ed_exalted_dynast
	has_game_rule = ed_rule_strip_on_leave_house_on
	NAND = {
		ed_character_house_is_exalted = yes
	}
}

ed_should_skip_renown_for_sovereign = {
	has_game_rule = ed_rule_stack_with_sovereign_skip_renown_on_sovereign
	ed_character_is_supreme_sovereign = yes
}

}

# Afford/rank triggers used by hub (rank checked in effect too)
ed_house_legacy_fertility_under_max = {
	exists = house
	house = {
		OR = {
			NOT = { has_variable = ed_legacy_fertility }
			var:ed_legacy_fertility < 3
		}
	}
}

ed_house_legacy_knight_under_max = {
	exists = house
	house = {
		OR = {
			NOT = { has_variable = ed_legacy_knight }
			var:ed_legacy_knight < 3
		}
	}
}

ed_house_legacy_prestige_under_max = {
	exists = house
	house = {
		OR = {
			NOT = { has_variable = ed_legacy_prestige }
			var:ed_legacy_prestige < 3
		}
	}
}

ed_dynasty_can_afford_legacy_rank_1 = {
	exists = dynasty
	dynasty = { dynasty_prestige >= 250 }

	exists = dynasty
	dynasty = { dynasty_prestige >= 500 }
}

ed_dynasty_can_afford_legacy_rank_3 = {

	dynasty = { dynasty_prestige >= 1000 }
}

	ed_legacy_track_enabled = yes
}

### Next-iter — secondary exalt

ed_actor_heads_primary_exalted_house = {
	is_alive = yes
	is_ai = no
	exists = house
	is_house_head = yes
	ed_character_house_is_primary_exalted = yes
}

ed_under_secondary_cap = {
	# Prefer dynasty-tracked count; treat missing as 0
	exists = dynasty
	OR = {
		AND = {
			dynasty = { NOT = { has_variable = ed_secondary_exalt_count } }
			ed_secondary_cap_value > 0
		}
		dynasty = {
			has_variable = ed_secondary_exalt_count
			var:ed_secondary_exalt_count < ed_secondary_cap_value
		}
	}
}

ed_actor_may_bless_secondary = {
	ed_actor_heads_primary_exalted_house = yes
	ed_actor_may_exalt_by_granting_rule = yes
	NOT = { has_game_rule = ed_rule_secondary_cap_0 }
	custom_tooltip = {
		text = ed_tt_under_secondary_cap
		ed_under_secondary_cap = yes
	}
}

ed_recipient_house_may_be_blessed = {

	NOT = { ed_character_house_is_exalted = yes }
}
