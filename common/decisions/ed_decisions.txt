ed_exalt_my_house_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}
	decision_group_type = major
	desc = ed_exalt_my_house_decision_desc
	selection_tooltip = ed_exalt_my_house_decision_tooltip
	confirm_text = ed_exalt_my_house_decision_confirm

	ai_check_interval = 0

	is_shown = {
		is_alive = yes
		exists = house
		NOT = { has_game_rule = ed_rule_granting_disabled }
		OR = {
			is_ai = no
			has_game_rule = ed_rule_granting_player_and_ai
		}
		NOT = { ed_character_house_is_exalted = yes }
	}

	is_valid = {
		custom_tooltip = {
			text = ed_tt_actor_may_exalt
			ed_actor_may_exalt = yes
		}
	}

	effect = {
		custom_tooltip = ed_exalt_my_house_decision_effect_tt
		ed_exalt_house_effect = yes
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
